%% especime-fontes.tex
%% Espécime visual — Lora, Luciole e NEWJUNE nas situações reais da linha.
%%
%% Uso:
%%   cd crpsp-book/desenvolvimento/medida && lualatex especime-fontes.tex
%%
%% Artefato de julgamento visual: não passa em teste automático e fica fora
%% do verificar.sh. Serve para decidir, olhando, o que medidas-fontes.tsv
%% só descreve.
%%
%% Cada página traz a mancha, o corpo e a página REAIS em disputa — A5 para
%% cartilha, A4 para manual e relatorio — com as três situações que o
%% briefing de 16/09/2026 pede:
%%   1. título de capítulo sobre parágrafo  (\Huge\bfseries\sffamily)
%%   2. título de caixa em corpo pequeno    (\sffamily\footnotesize\bfseries, book:755)
%%   3. cabeçalho em corpo pequeno          (\Light\small, book:321-324)
%%
%% O MESMO conteúdo em todas as páginas: o que varia é quanta altura ele
%% ocupa, medida e impressa na legenda. É assim que se lê o custo de
%% paginação de 11 pt -> 12 pt sem repaginar nada de verdade.
%%
%% A diagramação é feita no nível baixo (\pagewidth, \hoffset) e NÃO com
%% geometry: \newgeometry não aceita mudança de tamanho de papel, e o
%% resultado é mancha de A4 desenhada em folha A5.

\documentclass{article}
\usepackage{fontspec}
\usepackage{xcolor}

\definecolor{crp1}{RGB}{4, 88, 110}
\definecolor{nota}{RGB}{110, 110, 110}

% ⚠️ Fonte com duas cópias instaladas não se resolve por nome NEM por
% nome de arquivo: em 17/09/2026 havia duas Atkinson Hyperlegible Next
% nesta máquina (TeX Live e pacote do Fedora), com cobertura diferente.
% O que estiver numa árvore TEXMF é fixado pelo kpathsea; o que vem do
% Nextcloud por fontconfig (Lora, NEWJUNE) não tem caminho estável e
% fica sem Path. Quem acrescentar a Atkinson a este espécime: fixe-a,
% como em medir-fontes.tex.
\edef\LucioleDir{%
  \directlua{
    local achado = kpse.find_file("Luciole-Regular.ttf", "truetype fonts")
    if achado then tex.sprint(-2, (achado:gsub("[^/]*$", ""))) end
  }%
}

\newfontfamily\fLora{Lora}
\newfontfamily\fLuciole{Luciole}[Path = \LucioleDir,
                            UprightFont = Luciole-Regular.ttf]
\newfontfamily\fNJ{NEWJUNE}[UprightFont = NEWJUNE-REGULAR.OTF,
                            BoldFont    = NEWJUNE-BOLD.OTF]
\newfontfamily\fNJLight{NEWJUNE-Light}[UprightFont = NEWJUNE-LIGHT.OTF]

\newcommand{\Corrido}{%
A psicóloga inscrita no Conselho Regional exerce função pública e está sujeita
à fiscalização ética. O órgão zela pela qualidade técnica do exercício
profissional, pela justiça das decisões e pela inclusão das pessoas atendidas,
em todo o território da sexta região.\par
Três princípios sustentam a atuação: o compromisso com os direitos humanos, a
defesa da ciência psicológica e a construção coletiva das orientações. Não há
exercício profissional sem responsabilidade, nem avaliação séria sem informação
clara. A câmara propôs questões de método que seguem em discussão.\par}

% ── Uma página de espécime ──────────────────────────────────────────
% #1 largura papel  #2 altura papel  #3 mancha  #4 corpo  #5 entrelinha
% #6 família de texto  #7 legenda
% O fator opcional multiplica a entrelinha para obter o espaço entre
% parágrafos (WCAG 1.4.8 pede >= 1,5). Passar 0 mostra o estado atual de
% uma linha que ainda não cumpre o critério.
\NewDocumentCommand{\Pagina}{O{1.5} m m m m m m m}{%
  \pagewidth=#2 \pageheight=#3
  \setlength{\textwidth}{#4}%
  \setlength{\textheight}{\dimexpr#3-2\margemv\relax}%
  \setlength{\hoffset}{\dimexpr(#2-#4)/2-1in\relax}%
  \setlength{\voffset}{\dimexpr\margemv-1in\relax}%
  #7\fontsize{#5}{#6}\selectfont
  % 3. cabeçalho em corpo pequeno
  {\fNJLight\small\color{crp1}\noindent Espécime CRP-SP\hfill\thepage\par}%
  \vspace{1mm}{\color{crp1}\hrule height 0.4pt}\vspace{6mm}%
  % 1. título de capítulo sobre parágrafo
  {\fNJ\Huge\bfseries\color{crp1}\noindent Fiscalização\par}%
  \vspace{4mm}%
  % Corpo medido: o mesmo texto em toda página, a altura é que varia.
  \begin{lrbox}{\corpobox}%
    \begin{minipage}{\textwidth}%
      #7\fontsize{#5}{#6}\selectfont
      % Espaço entre parágrafos do WCAG 1.4.8, e sem recuo: com parágrafo
      % em bloco o recuo é redundante.
      \setlength{\parindent}{0pt}%
      \setlength{\parskip}{#1\baselineskip}%
      \Corrido
      % 2. título de caixa em corpo pequeno
      \noindent\fbox{\parbox{\dimexpr\textwidth-2\fboxsep-2\fboxrule}{%
        \setlength{\parskip}{0pt}%
        {\fNJ\footnotesize\bfseries\color{crp1}Nota técnica\par}%
        \vspace{1mm}#7\footnotesize
        O prazo de manifestação corre a partir da ciência do interessado, na
        forma do regimento. A contagem exclui o dia do começo e inclui o do
        vencimento.\par}}\par
    \end{minipage}%
  \end{lrbox}%
  \setlength{\corpoalt}{\dimexpr\ht\corpobox+\dp\corpobox\relax}%
  \noindent\usebox{\corpobox}%
  \vfill
  {\fNJLight\scriptsize\color{nota}\noindent #8\\
   Altura do bloco de corpo: \the\corpoalt\ numa mancha vertical de
   \the\textheight.\par}%
  \clearpage
}
