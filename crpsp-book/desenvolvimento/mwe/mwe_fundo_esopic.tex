% Achado do teste 1 (2026-07-13): includegraphics com alt={} sob
% tagging vira <Figure> com Alt vazio — veraPDF UA-2 até PASSA, mas
% o requisito do briefing é ARTIFACT (fundo fora da árvore de tags).
% Portanto: \tagstop suspende a criação de estrutura (sem <Figure>)
% e \tagmcbegin{artifact} marca o conteúdo do stream como artefato
% (sem ele o conteúdo ficaria sem MC — falha "untagged content").
\AddToShipoutPictureBG{%
    \tagstop
    \tagmcbegin{artifact}%
    \AtPageLowerLeft{%
        \ifodd\value{page}
            \includegraphics[width=\paperwidth,
                             height=\paperheight]{fundo_impar.png}%
        \else
            \includegraphics[width=\paperwidth,
                             height=\paperheight]{fundo_par.png}%
        \fi
    }%
    \tagmcend
    \tagstart
}
