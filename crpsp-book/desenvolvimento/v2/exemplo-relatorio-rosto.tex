%%   ~/verapdf/verapdf --format text --flavour ua2 exemplo-relatorio-rosto.pdf

%%     exemplo-relatorio-rosto.pdf | grep -cE '─Figure \('   # tem de dar 1

\documentclass[externo]{crpsp-relatorio}

  pdftitle  = {Exemplo de Folha de Rosto CRP-SP},
